# Re-enable only the H.264/HEVC/AV1 decoders and libdav1d wrapper.
# Native AV1 decoding uses D3D11VA/VideoToolbox hardware acceleration;
# get_pixel_format returns ENOSYS without usable hardware, so libdav1d handles
# software AV1 fallback. The client splits AV1 temporal units without a parser.
# scripts/build.py supplies platform-specific hardware accelerators to avoid
# configuring Windows-only d3d11va on Linux.
ffmpeg/*:disable_everything=True
ffmpeg/*:enable_decoders=h264,hevc,av1,libdav1d
ffmpeg/*:enable_parsers=h264,hevc
